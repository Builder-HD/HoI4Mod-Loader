# HOI4 Loader Conversion Rules: the equipment conversion game rules, in their own "Equipment
# Conversion" group. With the loader's equipment_conversion feature, a production line's "Convert
# from stockpile" also takes equipment of earlier generations (infantry_equipment_0 and _1 into
# _2, older chassis, airframes and hulls into newer ones). The loader reads the global flags that
# common/on_actions/hoi4_loader_conversion_rules_on_actions.txt sets from these at game start.
# Without the loader the rules show but do nothing.
hoi4_loader_conversion_generations = {
	name = "RULE_HOI4_LOADER_CONVERSION_GENERATIONS"
	desc = "RULE_HOI4_LOADER_CONVERSION_GENERATIONS_DESC"
	group = "RULE_GROUP_HOI4_LOADER_CONVERSION"
	default = {
		name = ALLOWED
		text = "RULE_OPTION_ALLOWED"
		desc = "RULE_HOI4_LOADER_CONVERSION_GENERATIONS_ALLOWED_DESC"
	}
	option = {
		name = BLOCKED
		text = "RULE_OPTION_BLOCKED"
		desc = "RULE_HOI4_LOADER_CONVERSION_GENERATIONS_BLOCKED_DESC"
		allow_achievements = yes
	}
}
