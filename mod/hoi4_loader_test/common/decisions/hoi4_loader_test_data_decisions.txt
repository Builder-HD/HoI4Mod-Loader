# AI-evaluated use for the cost measurement (loader_config.ini [diagnostics] trigger_timing = 1).
# Visible to every country while the features are active; the last modifier makes the AI never
# take it, so it only costs evaluation time.
hoi4_loader_ai_cost_category = {

	hoi4_loader_ai_cost_probe = {
		icon = generic_research
		cost = 0
		visible = {
			hoi4_loader_feature_army_condition = yes
			hoi4_loader_feature_military_counts = yes
			hoi4_loader_feature_region_control = yes
			hoi4_loader_feature_production_data = yes
		}
		ai_will_do = {
			base = 1
			modifier = { factor = 1 avg_division_strength < 0.5 }
			modifier = { factor = 1 avg_division_organisation < 0.5 }
			modifier = { factor = 1 battles_count = { involving = ROOT count > 0 } }
			modifier = { factor = 1 divisions_count = { region = 19 enemy_of = ROOT count > 0 } }
			modifier = { factor = 1 air_superiority = { region = 19 value > 0.5 } }
			modifier = { factor = 1 factories_assigned = { equipment = infantry_equipment count > 5 } }
			modifier = { factor = 0 always = yes }
		}
	}
}
