# The railway dismantling game rules as the global flags the loader reads. Game rules are fixed
# for a game, and global flags are saved with it, so setting them once at game start is enough.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { has_game_rule = { rule = hoi4_loader_railway_dismantle_allies option = ALLOWED } }
				set_global_flag = hoi4_loader_railway_dismantle_allies
			}
			if = {
				limit = { has_game_rule = { rule = hoi4_loader_railway_dismantle_subjects option = ALLOWED } }
				set_global_flag = hoi4_loader_railway_dismantle_subjects
			}
			if = {
				limit = { has_game_rule = { rule = hoi4_loader_railway_dismantle_overlord option = ALLOWED } }
				set_global_flag = hoi4_loader_railway_dismantle_overlord
			}
		}
	}
}
