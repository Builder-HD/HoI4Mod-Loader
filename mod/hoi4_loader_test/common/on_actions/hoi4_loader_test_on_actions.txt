on_actions = {
	# New game only: on_startup does NOT fire when a save is loaded.
	on_startup = {
		effect = {
			# on_startup effects run without a scope, and scripted effects need one,
			# so hoi4_loader_init is called from inside a country scope.
			every_country = {
				limit = { is_ai = no }
				# Diagnostics in logs/game.log.
				log = "hoi4_loader_test: on_startup fired"
				if = {
					limit = { has_global_flag = hoi4_loader_plugin_engine_probe }
					log = "hoi4_loader_test: engine_probe flag was SET before hoi4_loader_init"
				}
				else = {
					log = "hoi4_loader_test: engine_probe flag was NOT set before hoi4_loader_init"
				}
				hoi4_loader_init = yes
				# New games start with current data: nothing to migrate.
				set_variable = { global.hoi4_loader_test_mod_version = 2 }
				country_event = { id = hoi4_loader_test.1 hours = 1 }
			}
		}
	}
	# Loaded saves keep the flags they were saved with, which may not match this session's
	# loader (e.g. engine tier now disabled). Refresh them weekly; ROOT is each country.
	# The migration runs here too: the fallback for sessions where on_game_loaded does not fire
	# (no loader, multiplayer, feature switched off). It does nothing once migrated.
	on_weekly = {
		effect = {
			if = {
				limit = { is_ai = no }
				hoi4_loader_init = yes
				hoi4_loader_test_migrate = yes
			}
		}
	}
	# hoi4_loader: fires after a save is loaded (never for a new game), once per human-controlled
	# country with ROOT = that country, after the loader refreshed its flags and region names.
	# Single player only in this loader version. Without the loader it never fires (no error).
	on_game_loaded = {
		effect = {
			log = "hoi4_loader_test: on_game_loaded fired"
			add_to_variable = { global.hoi4_loader_test_loads = 1 }
			set_global_flag = hoi4_loader_test_in_on_game_loaded
			hoi4_loader_test_migrate = yes
			clr_global_flag = hoi4_loader_test_in_on_game_loaded
			country_event = { id = hoi4_loader_test.3 }
		}
	}
}
